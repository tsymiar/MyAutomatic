├── lookup/            # SQLite 多维度查找 (time/seek/tool)
├── media/             # FFmpeg 音视频处理
├── thread/            # 纯 C 线程池 (pthdpool)
├── trade/             # 量化交易策略 (Python)
├── webev/             # 事件驱动 Web 服务器 (epoll)
├── vs-extension/      # VSCode 文件夹比较扩展 (TS + Python, 相似度分析)
